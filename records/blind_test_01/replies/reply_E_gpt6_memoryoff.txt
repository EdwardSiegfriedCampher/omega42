from itertools import product
from collections import Counter
from math import gcd, isclose, pi
import cmath

N = 9
V = list(product(range(N), repeat=2))
I = (1, 0, 0, 1)

def Q(x):
    a, b = x
    return (a*a + a*b - b*b) % N

def act(A, x):
    a, b = x
    r, s, t, u = A
    return ((r*a + s*b) % N, (t*a + u*b) % N)

def det(A):
    r, s, t, u = A
    return (r*u - s*t) % N

def mm(A, B):
    a, b, c, d = A
    e, f, g, h = B
    return (
        (a*e + b*g) % N,
        (a*f + b*h) % N,
        (c*e + d*g) % N,
        (c*f + d*h) % N
    )

def mpow(A, n):
    out = I
    for _ in range(n):
        out = mm(out, A)
    return out

def order(A):
    out = I
    for n in range(1, 100):
        out = mm(out, A)
        if out == I:
            return n
    raise ValueError("Order not found")

# Enumerate GL(2, Z/9Z).
GL = [
    A for A in product(range(N), repeat=4)
    if gcd(det(A), N) == 1
]

# Test unitary and anti-isometric conditions on all 81 vectors.
O = [
    A for A in GL
    if all((Q(act(A, x)) - Q(x)) % N == 0 for x in V)
]
Anti = [
    A for A in GL
    if all((Q(act(A, x)) + Q(x)) % N == 0 for x in V)
]
G = set(O) | set(Anti)

print("Number of vectors:", len(V))

    Counter(order(A) for A in Anti).items()
)))

    Counter(order(A) for A in G).items()
