Setup

Let V=(\mathbb Z/9\mathbb Z)^2. For x=(a,b), write N(a,b)=a^2+ab-b^2, so q(x)=N(a,b)/9 \pmod 1. The polarization is \beta(x,y)=q(x+y)-q(x)-q(y) \pmod 1. The symmetric matrix of \beta (up to the 1/9 scaling) is

A=\begin{pmatrix}2&1\\1&-2\end{pmatrix},

since 2aa'+ab'+a'b-2bb'=(a,b)\,A\,(a',b')^\top.

All computations below are exact, over the finite set V of 81 elements. The code is Python 3 with only the standard library.

```python

def q(a,b): return Fraction(N(a,b), m)
```

```python
def mat_mul(M, N):
    return ((M[0]*N[0]+M[1]*N[2]) % m, (M[0]*N[1]+M[1]*N[3]) % m,
            (M[2]*N[0]+M[3]*N[2]) % m, (M[2]*N[1]+M[3]*N[3]) % m)

def act(M, x):
    return ((M[0]*x[0]+M[1]*x[1]) % m, (M[2]*x[0]+M[3]*x[1]) % m)

def det(M): return (M[0]*M[3] - M[1]*M[2]) % m
def is_unit(M): return math.gcd(det(M), m) == 1

def preserves(M, sign):
    for x in V:
        if N(*act(M, x)) != (sign * N(*x)) % m:
            return False
    return True

GL = [M for M in itertools.product(range(m), repeat=4) if is_unit(M)]
```
