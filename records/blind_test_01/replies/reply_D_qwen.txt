import numpy as np

from math import gcd
from collections import Counter

MOD = 9

def q(a, b):
    """q(a,b) = (a^2 + ab - b^2) / 9 mod 1, returned as numerator mod 9."""
    return (a*a + a*b - b*b) % MOD

def beta(x, y):
    """Bilinear form beta(x,y) = q(x+y) - q(x) - q(y) mod 9 (numerator)."""
    (a,b), (c,d) = x, y
    return (2*a*c + a*d + b*c - 2*b*d) % MOD

def mat_mul(A, B, mod):

def mat_transpose(A):
    return (A[0], A[2], A[1], A[3])  # (a,b,c,d) -> (a,c,b,d)

def mat_apply(A, x, mod):

def is_invertible(A, mod):
