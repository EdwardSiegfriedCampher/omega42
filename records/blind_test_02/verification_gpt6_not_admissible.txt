```python

from itertools import product
from math import gcd
from collections import Counter

N = 9
V = list(product(range(N), repeat=2))
I = (1, 0, 0, 1)  # row-major 2x2 identity

def det(A):
    a, b, c, d = A
    return (a*d - b*c) % N

def act(A, x):
    a, b, c, d = A
    u, v = x
    return ((a*u + b*v) % N,
            (c*u + d*v) % N)

def matmul(A, B):
    a, b, c, d = A
    e, f, g, h = B
    return ((a*e + b*g) % N,
            (a*f + b*h) % N,
            (c*e + d*g) % N,
            (c*f + d*h) % N)

def q(x):
    a, b = x
    return (a*a + b*b) % N

def order(A):
    P = I
    for k in range(1, 1000):
        P = matmul(P, A)
        if P == I:
            return k
    raise ValueError("Order not found")

def cycle_census(A):
    unseen = set(V)
    census = Counter()
    while unseen:
        start = min(unseen)
        x = start
        length = 0
        while True:
            unseen.remove(x)
            length += 1
            x = act(A, x)
            if x == start:
                break
        census[length] += 1
    return dict(sorted(census.items()))

all_invertible = []
preserving = []
reversing = []

for A in product(range(N), repeat=4):
    if gcd(det(A), N) != 1:
        continue

    all_invertible.append(A)
    preserves = all(q(act(A, x)) == q(x) for x in V)
    reverses = all(q(act(A, x)) == (-q(x)) % N
                   for x in V)

    if preserves:
        preserving.append(A)
    if reverses:
        reversing.append(A)

max_order = max(order(A) for A in reversing)
max_reversing = [
    A for A in reversing if order(A) == max_order
]

print("Invertible matrices:", len(all_invertible))
print("Spin-preserving:", len(preserving), preserving)
print("Spin-reversing:", len(reversing), reversing)
print("Maximum reversing order:", max_order)
print("Maximum-order reversing maps:")
for A in max_reversing:

    print({
        "matrix": A,
        "trace": (a+d) % N,
        "determinant": det(A),
        "order": order(A),
        "cycles": cycle_census(A),
        "A_squared": matmul(A, A),
        "quadratic_s": (a+d) % N,
        "quadratic_t": 1
```
